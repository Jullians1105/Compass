        // Micro-interactions for buttons
        document.querySelectorAll('button').forEach(btn => {
            btn.addEventListener('click', function(e) {
                if(this.innerText.includes('Presente') || this.innerText.includes('Ausente') || this.innerText.includes('Tardanza')) {
                    // Logic for toggle buttons
                    const container = this.parentElement;
                    container.querySelectorAll('button').forEach(b => {
                        b.classList.remove('bg-secondary/10', 'border-secondary', 'text-secondary', 'bg-error/10', 'border-error', 'text-error', 'bg-tertiary-fixed', 'border-tertiary-container', 'text-tertiary-container');
                        b.classList.add('border-outline', 'text-on-surface-variant');
                        const icon = b.querySelector('.material-symbols-outlined');
                        if(icon) icon.remove();
                    });

                    this.classList.remove('border-outline', 'text-on-surface-variant');
                    
                    if(this.innerText.trim() === 'Presente') {
                        this.classList.add('bg-secondary/10', 'border-secondary', 'text-secondary');
                        this.insertAdjacentHTML('afterbegin', '<span class="material-symbols-outlined text-sm" style="font-variation-settings: \'FILL\' 1;">check_circle</span>');
                    } else if(this.innerText.trim() === 'Ausente') {
                        this.classList.add('bg-error/10', 'border-error', 'text-error');
                        this.insertAdjacentHTML('afterbegin', '<span class="material-symbols-outlined text-sm" style="font-variation-settings: \'FILL\' 1;">cancel</span>');
                    } else if(this.innerText.trim() === 'Tardanza') {
                        this.classList.add('bg-tertiary-fixed', 'border-tertiary-container', 'text-tertiary-container');
                        this.insertAdjacentHTML('afterbegin', '<span class="material-symbols-outlined text-sm" style="font-variation-settings: \'FILL\' 1;">schedule</span>');
                    }
                }
            });
        });
